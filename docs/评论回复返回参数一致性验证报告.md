```javascript
handleReplyResultDatabase(data) {
    const { reply_id, request_id, status, platform_reply_id, error_code, error_message } = data;

    const reply = this.replyDAO.getReplyById(reply_id);
    if (!reply) {
        logger.warn(`Reply not found: ${reply_id}`);
        return;
    }

    // ✅ 成功：保存平台回复ID和数据
    if (status === 'success') {
        this.replyDAO.updateReplySuccess(reply_id, platform_reply_id, data.data);
        logger.info(`Reply success updated: ${reply_id}`, { platformReplyId: platform_reply_id });
    }

    // ❌ 失败：保存错误信息
    else if (status === 'failed' || status === 'blocked' || status === 'error') {
        this.replyDAO.updateReplyFailed(reply_id, error_code || 'UNKNOWN_ERROR', error_message || 'Unknown error');
        logger.info(`Reply failure updated: ${reply_id}`, { errorCode: error_code });
    }
}
```

```javascript
updateReplySuccess(replyId, platformReplyId, replyData = null) {
    const now = Date.now();
    const stmt = this.db.prepare(`
        UPDATE replies
        SET
            reply_status = 'success',
            platform_reply_id = ?,      -- ⚠️ 存储平台回复ID
            reply_data = ?,             -- 存储附加数据（JSON格式）

            updated_at = ?

    stmt.run(
        platformReplyId,                              // ⚠️ 关键参数
        replyData ? JSON.stringify(replyData) : null,
        now,
        now,
        replyId
    );
}
```

## 关键发现和建议

### ✅ 已验证的一致性

1. **字段命名一致**: `platform_reply_id` 在整个链路中保持一致
2. **数据类型一致**: 所有层级都使用 String 类型
3. **成功状态一致**: `success: true` → `status: 'success'` → `reply_status: 'success'`
4. **失败处理一致**: 错误码和错误消息完整传递
5. **附加数据保留**: 完整的评论信息保存到数据库

### ⚠️ 注意事项

1. **评论层级信息**:
   - ✅ 已包含在 `data.commentLevel` 中
   - ✅ 可用于统计分析（一级/二级/三级评论比例）

2. **时间戳格式**:
   - 视频详情页模块: ISO 8601 字符串 (`timestamp`)
   - Platform.js: Date 对象转 ISO 字符串
   - ReplyExecutor: Unix 时间戳 (`Date.now()`)
   - Master 数据库: Unix 时间戳

3. **可选字段处理**:
   - `platform_reply_id`: 成功时必需，失败时为 null
   - `error_code` / `error_message`: 失败时必需，成功时为 undefined

### 💡 优化建议

虽然当前实现完全一致，但可以考虑以下改进：

1. **类型定义**:
   ```javascript
   // 可以在 shared/types/ 中定义统一的类型
   interface ReplyResult {
       success: boolean;
       platform_reply_id?: string;
       data?: object;
       error?: string;
   }
   ```

2. **错误码枚举**:
   ```javascript
   // shared/constants/error-codes.js
   module.exports = {
       LOGIN_EXPIRED: 'LOGIN_EXPIRED',
       NETWORK_ERROR: 'NETWORK_ERROR',
       QUOTA_EXCEEDED: 'QUOTA_EXCEEDED',
       TARGET_NOT_FOUND: 'TARGET_NOT_FOUND',
       UNKNOWN_ERROR: 'UNKNOWN_ERROR',
   };
   ```

3. **日志追踪**:
   - 在每个层级添加 `trace_id`，便于跟踪单个回复的完整流程
